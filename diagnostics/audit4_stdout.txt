    24|   return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

    27| export function extractVoice(item: {
    28|   title?: string;
    29|   summary?: string;
    30|   source: string;
    31|   date: string;
    32|   relevance: number;
    33|   topic_relevance?: number;
    34|   axes: string[];
    35| }): ExtractedVoice[] {
    36|   // Criterion: minimum relevance AND speaker name in title/summary.
    37|   // Below title_only_below, speaker must be in title (not just summary) —
    38|   // avoids 'passing mention' false positives.
    39|   const relevanceForVoices = item.topic_relevance ?? item.relevance;
    40|   if (relevanceForVoices < voicesConfig.min_relevance) return [];
    41| 
    42|   const titleText = (item.title || '').toLowerCase();
    43|   const matches: ExtractedVoice[] = [];
    44| 
    45|   for (const voice of voicesConfig.voices) {
    46|     for (const keyword of voice.keywords) {
    47|       // Unicode-aware word boundaries: \b only works for [A-Za-z0-9_],
    48|       // fails for CJK and other non-Latin scripts. \p{L} matches any
    49|       // Unicode letter, /u enables Unicode mode.
    50|       const re = new RegExp(
    51|         `(?<!\\p{L})${escapeRegex(keyword.toLowerCase())}(?!\\p{L})`,
    52|         'u'
    53|       );
    54|       // Scientific integrity (§12): speaker must be in TITLE.
    55|       // Summary-only matches produced misleading attributions
    56|       // (e.g. "AMD acquires World Labs" -> Fei-Fei Li).
    57|       if (!re.test(titleText)) continue;
    58|       {
    59|         matches.push({
    60|           speaker: voice.name,
    61|           affiliation: (voice as any).affiliation ?? "",
    62|           category: voice.category,
    63|           quote: item.title || '',
    64|           date: item.date,
    65|           source: item.source,
    66|           relevance: item.relevance,
    67|           axes: item.axes,
    68|         });
    69|         break; // one quote per speaker per item
    70|       }
    71|     }
    72|   }
    73| 
    74|   return matches;
    75| }

  --- 0014_voices.sql ---
-- ============================================================
-- Migration 0014: voices table for automatic quote extraction
-- ============================================================
-- Stores quotes from curated speakers (see config/voices.yaml).
-- Populated automatically by Worker when items match criteria:
-- - relevance >= 0.8
-- - axes includes geopolitics OR h2_sovereignty
-- - speaker name in title/summary OR source in sources list
--
-- Uniqueness: UNIQUE(item_hash) prevents duplicate quotes.
-- Translations (quote_ru, quote_zh) added by translate-protocols workflow.
--
-- Rollback: DROP TABLE voices;
-- ============================================================

CREATE TABLE IF NOT EXISTS voices (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    item_hash TEXT NOT NULL,
    speaker TEXT NOT NULL,
    affiliation TEXT,
    category TEXT NOT NULL,
    quote TEXT NOT NULL,
    quote_ru TEXT,
    quote_zh TEXT,
    date TEXT NOT NULL,
    source TEXT NOT NULL,
    relevance REAL NOT NULL,
    axes TEXT NOT NULL,
    created_at TEXT NOT NULL,
    UNIQUE(item_hash)
);

CREATE INDEX IF NOT EXISTS idx_voices_date ON voices(date);
CREATE INDEX IF NOT EXISTS idx_voices_category ON voices(category);

  --- 0015_voices_unique_item_speaker.sql ---
-- ============================================================
-- Migration 0015: voices UNIQUE(item_hash, speaker)
-- ============================================================
-- Previous UNIQUE(item_hash) allowed only one quote per item.
-- When multiple curated speakers appeared in the same item
-- (e.g. "Inside Zuckerberg, Huang's push..."), only the first
-- match was saved. New UNIQUE allows one quote per (item, speaker).
--
-- Context: 2026-10-07, fixed word-boundary extraction + multi-speaker.
--
-- Rollback: revert to 0014 schema (UNIQUE(item_hash) only).
-- ============================================================

DROP TABLE IF EXISTS voices;

CREATE TABLE voices (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    item_hash TEXT NOT NULL,
    speaker TEXT NOT NULL,
    affiliation TEXT,
    category TEXT NOT NULL,
    quote TEXT NOT NULL,
    quote_ru TEXT,
    quote_zh TEXT,
    date TEXT NOT NULL,
    source TEXT NOT NULL,
    relevance REAL NOT NULL,
    axes TEXT NOT NULL,
    created_at TEXT NOT NULL,
    UNIQUE(item_hash, speaker)
);

CREATE INDEX IF NOT EXISTS idx_voices_date ON voices(date);
CREATE INDEX IF NOT EXISTS idx_voices_category ON voices(category);
